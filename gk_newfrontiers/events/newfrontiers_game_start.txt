namespace = newfrontiers

# Flag empires whose founder started on Sterile. Gates who can gene-mod sterile
# habitability and terraform worlds into Sterile.
event = {
	id = newfrontiers.111
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		every_country = {
			limit = {
				NOT = { has_country_flag = gk_sterile_dweller }
				OR = {
					founder_species = {
						OR = {
							has_trait = trait_pc_sterile_preference
							has_trait = trait_sterile_planet_preference
						}
					}
					capital_scope = { is_planet_class = pc_sterile }
				}
			}
			set_country_flag = gk_sterile_dweller
		}
	}
}

### Generate Starting Subclass
event = {
	id = newfrontiers.100
	hide_window = yes
	is_triggered_only = yes
	
	immediate = {
		# Start Subclass Menu
		every_country = {
			limit = { is_ai = no }
			country_event = { id = newfrontiers.105 }
		}
		
		every_country = {
			limit = { 
				is_ai = yes
				OR = {
					is_country_type = default
					#is_country_type = fallen_empire
					is_country_type = primitive
				}
			}					
			capital_scope = { carrier_event = { id = newfrontiers.109 } }
		}
		
		#Living Sea
		random_galaxy_planet = {
			limit = { has_modifier = "living_sea" }
			set_planet_entity = {
				picture = pc_living_sea
				entity = ocean_planet_sea_of_conciousness_entity
			}
		}
	}
}

	id = newfrontiers.104
	hide_window = yes
	is_triggered_only = yes
	
	immediate = {
		every_galaxy_planet = {	
			
			limit = {
				AND = {
					OR = {
						solar_system = {
							OR = {
								is_star_class = sc_t
							}
						}
						AND = {
							solar_system = {
								is_star_class = sc_m
							}
							has_carrier_flag = cold_zone
						}
					}
					OR = {
						is_planet_class = "pc_desert"
						is_planet_class = "pc_arid"
						is_planet_class = "pc_ocean"
						is_planet_class = "pc_tundra"
						is_planet_class = "pc_arctic"
						is_planet_class = "pc_alpine"
					}
					NOT = { num_modifiers > 2 }
					NOT = { has_modifier = "long_night" }
				}
			}
			add_modifier = {
				modifier = "eternal_night"
				days = -1
			}
		}
	}
