# Automodding delta for pc_sterile, the fourteenth sibling of vanilla's
# trait_auto_* set. Copied from trait_auto_pc_volcanic_preference with sterile
# taking the bonus and volcanic dropping to the penalty every other foreign
# class gets. Without this, pc_sterile's auto_trait_prio would have to point at
# trait_auto_dry_preference, which penalises Sterile on its own world.
trait_auto_pc_sterile_preference = {
	inline_script = {
		script = traits/auto_preference_planet_class
		PLANET_CLASS = pc_sterile
	}
	modifier = {
		# Dry
		pc_desert_habitability 			= @auto_penalty
		pc_arid_habitability 			= @auto_penalty
		pc_savannah_habitability 		= @auto_penalty
		# Wet
		pc_tropical_habitability 		= @auto_penalty
		pc_continental_habitability 	= @auto_penalty
		pc_ocean_habitability 			= @auto_penalty
		# Cold
		pc_tundra_habitability 			= @auto_penalty
		pc_arctic_habitability 			= @auto_penalty
		pc_alpine_habitability 			= @auto_penalty
		# Special
		pc_nuked_habitability 			= @auto_penalty
		pc_volcanic_habitability 		= @auto_penalty
		pc_sterile_habitability 		= @auto_bonus
	}
}
